// ================================================================
//  Firestore 보안 규칙 "추가분"
//  사회 게임 RPG에서 쓰던 규칙을 지우지 말고,
//  기존 규칙의  match /databases/{database}/documents { ... }  안쪽에
//  아래 블록만 붙여 넣은 뒤 [게시]를 누르세요.
// ================================================================

    // 경제 수업 게임 (경제 실험실 · 금융 투자 타임머신)
    match /econRooms/{roomId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null
                    && request.resource.data.host == request.auth.uid;
      allow update, delete: if request.auth != null
                    && resource.data.host == request.auth.uid;

      match /players/{pid} {
        allow read: if request.auth != null;
        // 학생은 자기 정보만, 선생님(방 주인)은 모든 학생 정보를 쓸 수 있음
        allow write: if request.auth != null && (
          request.auth.uid == pid ||
          get(/databases/$(database)/documents/econRooms/$(roomId)).data.host == request.auth.uid
        );
      }
    }
